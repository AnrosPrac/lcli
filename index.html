#!/bin/bash D="$HOME/.lum" RAW="https://raw.githubusercontent.com/AnrosPrac/lcli/main" echo "[*] Installing Lum CLI dependencies..." # 1. Install Python dependencies (correct set) python3 -m pip install --user \ httpx \ websockets \ pynacl \ -q --no-warn-script-location --disable-pip-version-check # 2. Create directory & download CLI mkdir -p "$D" curl -sSL "$RAW/lum.py" -o "$D/lum.py" # 3. Add alias to shell config (bash + zsh) A="alias lum='python3 $D/lum.py'" for rc in "$HOME/.bashrc" "$HOME/.zshrc"; do if [ -f "$rc" ] && ! grep -qF "$A" "$rc"; then echo "$A" >> "$rc" fi done # 4. Activate alias for current session eval "$A" echo "" echo "[✔] Lum CLI installed successfully" echo "[!] Please close and reopen your terminal to use the 'lum' command"
